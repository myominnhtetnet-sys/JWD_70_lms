
package com.Learning_Managnment_System.JWD_70_lms.Controller;

import java.util.List;
import java.util.Map;

import jakarta.servlet.http.HttpSession;

import org.springframework.stereotype.Controller;
import org.springframework.ui.Model;
import org.springframework.validation.BindingResult;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.ModelAttribute;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RequestParam;
import org.springframework.web.servlet.mvc.support.RedirectAttributes;

import com.Learning_Managnment_System.JWD_70_lms.Service.TeacherExamResultService;
import com.Learning_Managnment_System.JWD_70_lms.model.ExamGradingForm;

@Controller
@RequestMapping("/teacher/exam-results")
public class TeacherExamResultController {

    private final TeacherExamResultService service;

    public TeacherExamResultController(
            TeacherExamResultService service) {
        this.service = service;
    }

    @GetMapping
    public String listResults(
            @RequestParam(name = "attemptId", required = false)
            Long attemptId,
            @RequestParam(name = "mode", defaultValue = "view")
            String mode,
            HttpSession session,
            Model model,
            RedirectAttributes redirectAttributes) {

        Long teacherId = getLoggedInTeacherId(session);

        if (teacherId == null) {
            return "redirect:/login";
        }

        loadList(model, teacherId);

        if (attemptId != null) {
            try {
                Map<String, Object> attempt =
                        service.getAttempt(attemptId, teacherId);

                List<Map<String, Object>> answers =
                        service.getAnswers(attempt);

                model.addAttribute("attempt", attempt);
                model.addAttribute("examAnswers", answers);
                model.addAttribute(
                        "gradingForm",
                        service.createForm(attempt, answers));

                model.addAttribute(
                        "mode",
                        "grade".equals(mode) ? "grade" : "view");

            } catch (IllegalArgumentException e) {
                redirectAttributes.addFlashAttribute(
                        "error", e.getMessage());

                return "redirect:/teacher/exam-results";
            }
        }

        return "teacher-exam-result-list";
    }

    @PostMapping("/{attemptId}/grade")
    public String saveGrades(
            @PathVariable("attemptId") Long attemptId,
            @ModelAttribute("gradingForm") ExamGradingForm form,
            BindingResult bindingResult,
            HttpSession session,
            Model model,
            RedirectAttributes redirectAttributes) {

        Long teacherId = getLoggedInTeacherId(session);

        if (teacherId == null) {
            return "redirect:/login";
        }

        Map<String, Object> attempt;

        try {
            attempt = service.getAttempt(attemptId, teacherId);

        } catch (IllegalArgumentException e) {
            redirectAttributes.addFlashAttribute(
                    "error", e.getMessage());

            return "redirect:/teacher/exam-results";
        }

        List<Map<String, Object>> answers =
                service.getAnswers(attempt);

        try {
            if (bindingResult.hasErrors()) {
                throw new IllegalArgumentException(
                        "Please enter valid numbers for the marks.");
            }

            service.gradeAttempt(attemptId, teacherId, form);

            redirectAttributes.addFlashAttribute(
                    "success", "Exam graded successfully!");

            return "redirect:/teacher/exam-results";

        } catch (IllegalArgumentException e) {
            loadList(model, teacherId);

            model.addAttribute("attempt", attempt);
            model.addAttribute("examAnswers", answers);
            model.addAttribute("mode", "grade");
            model.addAttribute("error", e.getMessage());

            return "teacher-exam-result-list";
        }
    }

    private Long getLoggedInTeacherId(HttpSession session) {

        Object emailObject = session.getAttribute("email");
        Object roleObject = session.getAttribute("roleId");

        if (!(emailObject instanceof String email)
                || email.isBlank()) {
            return null;
        }

        if (!(roleObject instanceof Number role)
                || role.intValue() != 2) {
            return null;
        }

        try {
            return service.getTeacherId(email);
        } catch (IllegalArgumentException e) {
            return null;
        }
    }

    private void loadList(Model model, Long teacherId) {

        List<Map<String, Object>> results =
                service.getResults(teacherId);

        long graded = results.stream()
                .filter(row ->
                        "GRADED".equals(row.get("status")))
                .count();

        model.addAttribute("results", results);
        model.addAttribute("totalCount", results.size());
        model.addAttribute("gradedCount", graded);
        model.addAttribute(
                "pendingCount", results.size() - graded);
    }
}

package com.Learning_Managnment_System.JWD_70_lms.Repository;

import java.math.BigDecimal;
import java.util.List;
import java.util.Map;
import java.util.Optional;

import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.stereotype.Repository;

@Repository
public class TeacherExamResultRepository {

    private final JdbcTemplate jdbcTemplate;

    public TeacherExamResultRepository(JdbcTemplate jdbcTemplate) {
        this.jdbcTemplate = jdbcTemplate;
    }

    public Optional<Long> findUserIdByEmail(String email) {

        List<Long> ids = jdbcTemplate.query(
                "SELECT user_id FROM users WHERE email = ?",
                (rs, rowNum) -> rs.getLong("user_id"),
                email);

        return ids.stream().findFirst();
    }

    public List<Map<String, Object>> findResults(Long teacherId) {

        String sql = """
                SELECT
                    ea.attempt_id AS attemptId,
                    ea.exam_id AS examId,
                    ea.attempt_no AS attemptNo,
                    ea.submitted_at AS submittedAt,
                    ea.score AS score,
                    ea.result AS result,
                    ea.status AS status,
                    ex.title AS examTitle,
                    ex.total_mark AS totalMark,
                    u.full_name AS studentName
                FROM exam_attempts ea
                JOIN exams ex
                    ON ex.exam_id = ea.exam_id
                JOIN enrollments en
                    ON en.enrollment_id = ea.enrollment_id
                JOIN users u
                    ON u.user_id = en.user_id
                WHERE ex.created_by = ?
                  AND ea.status IN ('SUBMITTED', 'GRADED')
                ORDER BY ea.attempt_id DESC
                """;

        return jdbcTemplate.queryForList(sql, teacherId);
    }

    public Optional<Map<String, Object>> findAttempt(
            Long attemptId,
            Long teacherId,
            boolean lock) {

        String sql = """
                SELECT
                    ea.attempt_id AS attemptId,
                    ea.exam_id AS examId,
                    ea.attempt_no AS attemptNo,
                    ea.submitted_at AS submittedAt,
                    ea.score AS score,
                    ea.result AS result,
                    ea.status AS status,
                    ea.remark AS remark,
                    ex.title AS examTitle,
                    ex.total_mark AS totalMark,
                    ex.pass_mark AS passMark,
                    u.full_name AS studentName
                FROM exam_attempts ea
                JOIN exams ex
                    ON ex.exam_id = ea.exam_id
                JOIN enrollments en
                    ON en.enrollment_id = ea.enrollment_id
                JOIN users u
                    ON u.user_id = en.user_id
                WHERE ea.attempt_id = ?
                  AND ex.created_by = ?
                """;

        if (lock) {
            sql += " FOR UPDATE";
        }

        List<Map<String, Object>> rows = jdbcTemplate.queryForList(
                sql, attemptId, teacherId);

        return rows.stream().findFirst();
    }

    public List<Map<String, Object>> findAnswers(
            Long attemptId,
            Long examId) {

        String sql = """
                SELECT
                    eq.question_id AS questionId,
                    eq.sort_order AS sortOrder,
                    eq.mark AS maxMark,
                    q.question_text AS questionText,
                    q.question_type AS questionType,
                    q.explanation AS explanation,
                    aa.answer_text AS answerText,
                    aa.mark_awarded AS markAwarded,
                    aa.selected_option_id AS selectedOptionId,
                    qo.option_text AS selectedOptionText,
                    qo.is_correct AS optionCorrect
                FROM exam_questions eq
                JOIN questions q
                    ON q.question_id = eq.question_id
                LEFT JOIN attempt_answers aa
                    ON aa.question_id = eq.question_id
                   AND aa.attempt_id = ?
                LEFT JOIN question_options qo
                    ON qo.option_id = aa.selected_option_id
                   AND qo.question_id = eq.question_id
                WHERE eq.exam_id = ?
                ORDER BY eq.sort_order, eq.question_id
                """;

        return jdbcTemplate.queryForList(sql, attemptId, examId);
    }

    public void saveAnswerMark(
            Long attemptId,
            Long questionId,
            BigDecimal mark,
            Boolean correct) {

        /*
         * Missing answer rows are also handled.
         * Existing student answer_text and selected_option_id are preserved.
         */
        String sql = """
                INSERT INTO attempt_answers
                    (attempt_id, question_id, mark_awarded, is_correct)
                VALUES (?, ?, ?, ?)
                ON DUPLICATE KEY UPDATE
                    mark_awarded = ?,
                    is_correct = ?
                """;

        jdbcTemplate.update(
                sql,
                attemptId,
                questionId,
                mark,
                correct,
                mark,
                correct);
    }

    public int completeGrading(
            Long attemptId,
            BigDecimal score,
            String result,
            String remark) {

        String sql = """
                UPDATE exam_attempts
                SET score = ?,
                    result = ?,
                    status = 'GRADED',
                    remark = ?,
                    updated_at = CURRENT_TIMESTAMP
                WHERE attempt_id = ?
                  AND status IN ('SUBMITTED', 'GRADED')
                """;

        return jdbcTemplate.update(
                sql, score, result, remark, attemptId);
    }
}

package com.Learning_Managnment_System.JWD_70_lms.Service;

import java.math.BigDecimal;
import java.math.RoundingMode;
import java.util.HashMap;
import java.util.HashSet;
import java.util.List;
import java.util.Map;
import java.util.Set;

import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import com.Learning_Managnment_System.JWD_70_lms.Repository.TeacherExamResultRepository;
import com.Learning_Managnment_System.JWD_70_lms.model.ExamGradingForm;
import com.Learning_Managnment_System.JWD_70_lms.model.ExamGradingForm.AnswerMark;

@Service
public class TeacherExamResultService {

    private final TeacherExamResultRepository repository;

    public TeacherExamResultService(
            TeacherExamResultRepository repository) {
        this.repository = repository;
    }

    public Long getTeacherId(String email) {
        return repository.findUserIdByEmail(email)
                .orElseThrow(() ->
                        new IllegalArgumentException(
                                "Logged-in user not found."));
    }

    public List<Map<String, Object>> getResults(Long teacherId) {
        return repository.findResults(teacherId);
    }

    public Map<String, Object> getAttempt(
            Long attemptId,
            Long teacherId) {

        Map<String, Object> attempt = repository
                .findAttempt(attemptId, teacherId, false)
                .orElseThrow(() ->
                        new IllegalArgumentException(
                                "Exam attempt not found or access denied."));

        checkStatus(attempt);
        return attempt;
    }

    public List<Map<String, Object>> getAnswers(
            Map<String, Object> attempt) {

        return repository.findAnswers(
                id(attempt.get("attemptId")),
                id(attempt.get("examId")));
    }

    public ExamGradingForm createForm(
            Map<String, Object> attempt,
            List<Map<String, Object>> answers) {

        ExamGradingForm form = new ExamGradingForm();
        form.setRemark((String) attempt.get("remark"));

        for (Map<String, Object> answer : answers) {

            if ("SHORT_ANSWER".equals(answer.get("questionType"))) {

                AnswerMark item = new AnswerMark();
                item.setQuestionId(id(answer.get("questionId")));
                item.setMark((BigDecimal) answer.get("markAwarded"));

                form.getAnswers().add(item);
            }
        }

        return form;
    }

    @Transactional
    public void gradeAttempt(
            Long attemptId,
            Long teacherId,
            ExamGradingForm form) {

        /*
         * Lock the attempt so two grading requests cannot update its
         * answers and total at the same time.
         */
        Map<String, Object> attempt = repository
                .findAttempt(attemptId, teacherId, true)
                .orElseThrow(() ->
                        new IllegalArgumentException(
                                "Exam attempt not found or access denied."));

        checkStatus(attempt);

        List<Map<String, Object>> answers = getAnswers(attempt);

        if (answers.isEmpty()) {
            throw new IllegalArgumentException(
                    "This exam has no questions.");
        }

        BigDecimal totalMark = (BigDecimal) attempt.get("totalMark");
        BigDecimal passMark = (BigDecimal) attempt.get("passMark");

        if (totalMark == null || totalMark.signum() <= 0
                || passMark == null || passMark.signum() < 0
                || passMark.compareTo(totalMark) > 0) {

            throw new IllegalArgumentException(
                    "Exam total mark or pass mark is invalid.");
        }

        Map<Long, BigDecimal> submittedMarks = new HashMap<>();

        if (form.getAnswers() == null) {
            throw new IllegalArgumentException("Invalid grading form.");
        }

        for (AnswerMark item : form.getAnswers()) {

            if (item == null || item.getQuestionId() == null) {
                throw new IllegalArgumentException(
                        "Invalid question in grading form.");
            }

            if (submittedMarks.containsKey(item.getQuestionId())) {
                throw new IllegalArgumentException(
                        "Duplicate question in grading form.");
            }

            submittedMarks.put(item.getQuestionId(), item.getMark());
        }

        Set<Long> shortQuestionIds = new HashSet<>();
        BigDecimal questionTotal = BigDecimal.ZERO;

        /*
         * Validate the complete form before writing any marks.
         */
        for (Map<String, Object> answer : answers) {

            BigDecimal maxMark = (BigDecimal) answer.get("maxMark");

            if (maxMark == null || maxMark.signum() <= 0) {
                throw new IllegalArgumentException(
                        "Question maximum mark must be greater than 0.");
            }

            questionTotal = questionTotal.add(maxMark);

            String type = (String) answer.get("questionType");

            if (!Set.of("MCQ", "TRUE_FALSE", "SHORT_ANSWER")
                    .contains(type)) {

                throw new IllegalArgumentException(
                        "Unsupported question type.");
            }

            if ("SHORT_ANSWER".equals(type)) {

                Long questionId = id(answer.get("questionId"));
                shortQuestionIds.add(questionId);

                BigDecimal mark = submittedMarks.get(questionId);

                if (mark == null) {
                    throw new IllegalArgumentException(
                            "Please grade every short answer.");
                }

                if (mark.signum() < 0 || mark.compareTo(maxMark) > 0) {
                    throw new IllegalArgumentException(
                            "Question " + questionId
                            + ": mark must be between 0 and "
                            + maxMark + ".");
                }

                if (mark.stripTrailingZeros().scale() > 2) {
                    throw new IllegalArgumentException(
                            "Marks can have at most 2 decimal places.");
                }
            }
        }

        if (!submittedMarks.keySet().equals(shortQuestionIds)) {
            throw new IllegalArgumentException(
                    "The grading form does not match this exam.");
        }

//        if (questionTotal.compareTo(totalMark) != 0) {
//            throw new IllegalArgumentException(
//                    "Question marks total (" + questionTotal
//                    + ") must match exam total mark (" + totalMark
//                    + "). Please fix the exam marks first.");
//        }

        if (questionTotal.signum() <= 0) {
            throw new IllegalArgumentException(
                    "Question marks total must be greater than 0.");
        }
        BigDecimal score = BigDecimal.ZERO;

        for (Map<String, Object> answer : answers) {

            Long questionId = id(answer.get("questionId"));
            String type = (String) answer.get("questionType");
            BigDecimal maxMark = (BigDecimal) answer.get("maxMark");

            BigDecimal awarded;
            Boolean correct;

            if ("SHORT_ANSWER".equals(type)) {

                awarded = submittedMarks.get(questionId);

                /*
                 * Partial credit does not have a single true/false
                 * meaning, so correctness remains NULL.
                 */
                correct = null;

            } else {

                correct = isTrue(answer.get("optionCorrect"));
                awarded = correct ? maxMark : BigDecimal.ZERO;
            }

            repository.saveAnswerMark(
                    attemptId, questionId, awarded, correct);

            score = score.add(awarded);
        }

//        String result = score.compareTo(passMark) >= 0
//                ? "PASSED"
//                : "FAILED";

     // မေးခွန်းအမှတ်ပေါင်းမှ Exam Total Mark အတိုင်း အချိုးကျပြောင်းမယ်
        score = score.multiply(totalMark)
                .divide(questionTotal, 2, RoundingMode.HALF_UP);

        // ပြောင်းပြီးသားအမှတ်ကို Exam Pass Mark နဲ့နှိုင်းယှဉ်မယ်
        String result = score.compareTo(passMark) >= 0
                ? "PASSED"
                : "FAILED";
        String remark = form.getRemark() == null
                ? ""
                : form.getRemark().trim();

        int updated = repository.completeGrading(
                attemptId, score, result, remark);

        if (updated != 1) {
            throw new IllegalArgumentException(
                    "Unable to save exam grading.");
        }
    }

    private void checkStatus(Map<String, Object> attempt) {

        String status = (String) attempt.get("status");

        if (!"SUBMITTED".equals(status) && !"GRADED".equals(status)) {
            throw new IllegalArgumentException(
                    "Only submitted or graded attempts can be reviewed.");
        }
    }

    private Long id(Object value) {
        return ((Number) value).longValue();
    }

    private boolean isTrue(Object value) {

        if (value instanceof Boolean bool) {
            return bool;
        }

        return value instanceof Number number
                && number.intValue() != 0;
    }
}

<script>
    const rows = Array.from(
        document.querySelectorAll(".result-row")
    );

    const searchInput = document.getElementById("searchInput");
    const examFilter = document.getElementById("examFilter");
    const statusFilter = document.getElementById("statusFilter");

    const exams = new Map();

    rows.forEach(row => {
        exams.set(row.dataset.examId, row.dataset.exam);
    });

    exams.forEach((title, id) => {
        examFilter.add(new Option(title, id));
    });

    function filterResults() {
        const keyword = searchInput.value.trim().toLowerCase();
        let visible = 0;

        rows.forEach(row => {
            const text = (
                (row.dataset.student || "") + " " +
                (row.dataset.exam || "")
            ).toLowerCase();

            const matches =
                text.includes(keyword) &&
                (!examFilter.value ||
                    row.dataset.examId === examFilter.value) &&
                (!statusFilter.value ||
                    row.dataset.status === statusFilter.value);

            row.hidden = !matches;

            if (matches) {
                visible++;
            }
        });

        document.getElementById("emptyRow").hidden = visible > 0;
    }

    searchInput.addEventListener("input", filterResults);
    examFilter.addEventListener("change", filterResults);
    statusFilter.addEventListener("change", filterResults);

    document.getElementById("resetButton")
        .addEventListener("click", () => {
            searchInput.value = "";
            examFilter.value = "";
            statusFilter.value = "";
            filterResults();
        });

    filterResults();

    const modal = document.getElementById("examModal");

    if (modal) {
        document.body.style.overflow = "hidden";

        const box = modal.querySelector(".modal-box");
        const closeLink = modal.querySelector(
            '[aria-label="Close modal"]'
        );

        box.focus();

        modal.addEventListener("click", event => {
            if (event.target === modal) {
                closeLink.click();
            }
        });

        modal.addEventListener("keydown", event => {
            if (event.key === "Escape") {
                closeLink.click();
            }

            if (event.key === "Tab") {
                const controls = Array.from(box.querySelectorAll(
                    'a[href], button, input:not([type="hidden"]), ' +
                    'select, textarea, [tabindex="0"]'
                )).filter(element => element.getClientRects().length);

                const first = controls[0];
                const last = controls[controls.length - 1];

                if (event.shiftKey &&
                    (document.activeElement === first ||
                     document.activeElement === box)) {
                    event.preventDefault();
                    last?.focus();
                } else if (!event.shiftKey &&
                    document.activeElement === last) {
                    event.preventDefault();
                    first?.focus();
                }
            }
        });
    }
</script>
