                const explanation =

                const question = {

                fetch(

                        body:
                            JSON.stringify(
                                question
                            )

                    }
                )

                .then(response => {

                    if (!response.ok) {

                        throw new Error(
                            'Failed to save question'
                        );

                    }

                    return response.json();

                })

                .then(questionId => {

                    console.log(
                        "Created Question ID:",
                        questionId
                    );


                    if (
                        questionType === 'MCQ' ||
                        questionType === 'TRUE_FALSE'
                    ) {

                        return saveOptions(
                            questionId
                        );

                    }

                    return null;

                })

                .then(() => {

                    alert(
                        'Question saved successfully!'
                    );

                    closeQuestionForm();

                    loadQuestions();

                })

                .catch(error => {

                    console.error(
                        'Error saving question:',
                        error
                    );

                    alert(
                        'Failed to save question.'
                    );

                });

            }
        );


    /*
     * =========================
     * Save Options
     * =========================
     */

    function saveOptions(questionId) {

        const questionType =
            document.getElementById(
                'questionType'
            ).value;


        let labels = [];


        if (questionType === 'MCQ') {

            labels =
                ['A', 'B', 'C', 'D'];

        }

        else if (
            questionType === 'TRUE_FALSE'
        ) {

            labels =
                ['A', 'B'];

        }


        const correctOption =
            document.querySelector(
                'input[name="correctOption"]:checked'
            );


        if (!correctOption) {

            throw new Error(
                'Please select correct answer'
            );

        }


        const correctLabel =
            correctOption.value;


        const requests =
            labels.map(label => {

                const optionText =
                    document.getElementById(
                        `option_${label}`
                    ).value;


                const option = {

                    optionLabel:
                        label,

                    optionText:
                        optionText,

                    isCorrect:
                        label === correctLabel

                };


                return fetch(
                    `/api/questions/${questionId}/options`,
                    {

                        method: 'POST',

                        headers: {

                            'Content-Type':
                                'application/json'

                        },

                        body:
                            JSON.stringify(
                                option
                            )

                    }
                )

                .then(response => {

                    if (!response.ok) {

                        throw new Error(
                            `Failed to save option ${label}`
                        );

                    }

                    return response.json();

                });

            });


        return Promise.all(
            requests
        );

    }

            'flex items-center gap-3 mb-3';


        optionDiv.innerHTML = `

            <input
                type="radio"
                name="editCorrectOption"
                value="${label}"
                class="w-4 h-4
                       text-indigo-600"
                required>


            <div class="w-10 h-10
                        flex items-center
                        justify-center
                        bg-white
                        border border-slate-200
                        rounded-xl
                        font-bold
                        text-slate-600">

                ${label}

            </div>


            <input
                type="text"
                id="editOption_${label}"
                value="${defaultText}"
                placeholder="Option ${label}"
                class="form-input"
                required>

        `;


        container.appendChild(
            optionDiv
        );

    }


    /*
     * =========================
     * Close Edit
     * =========================
     */

    function closeEditQuestion() {

        const modal =
            document.getElementById(
                'editQuestionModal'
            );

        modal.classList.add('hidden');
        modal.classList.remove('flex');


        document
            .getElementById(
                'editQuestionForm'
            )
            .reset();


        document
            .getElementById(
                'editOptionsContainer'
            )
            .innerHTML = '';


        document
            .getElementById(
                'editOptionsSection'
            )
            .classList.add(
                'hidden'
            );

    }


    /*
     * =========================
     * Edit Submit
     * =========================
     */

    document
        .getElementById(
            'editQuestionForm'
        )
        .addEventListener(
            'submit',
            function (event) {

                event.preventDefault();


                const questionId =
                    this.dataset.questionId;


                const courseId =
                    document.getElementById(
                        'editQuestionCourse'
                    ).value;


                const questionText =
                    document.getElementById(
                        'editQuestionText'
                    ).value;


                const questionType =
                    document.getElementById(
                        'editQuestionType'
                    ).value;


                const defaultMark =
                    document.getElementById(
                        'editDefaultMark'
                    ).value;


                const explanation =
                    document.getElementById(
                        'editExplanation'
                    ).value;


                const question = {

                    courseId:
                        Number(courseId),

                    questionText:
                        questionText,

                    questionType:
                        questionType,

                    defaultMark:
                        Number(defaultMark),

                    explanation:
                        explanation,

                    isActive:
                        true

                };


                fetch(
                    `/api/questions/${questionId}`,
                    {

                        method: 'PUT',

                        headers: {

                            'Content-Type':
                                'application/json'

                        },

                        body:
                            JSON.stringify(
                                question
                            )

                    }
                )

                .then(response => {

                    if (!response.ok) {

                        throw new Error(
                            'Failed to update question'
                        );

                    }

                    return response.json();

                })

                .then(result => {

                    if (
                        questionType === 'MCQ' ||
                        questionType === 'TRUE_FALSE'
                    ) {

                        return updateOptions(
                            questionId,
                            questionType
                        );

                    }

                    return null;

                })

                .then(() => {

                    alert(
                        'Question updated successfully!'
                    );

                    closeEditQuestion();

                    loadQuestions();

                })

                .catch(error => {

                    console.error(
                        'Error updating question:',
                        error
                    );

                    alert(
                        'Failed to update question.'
                    );

                });

            }
        );


    /*
     * =========================
     * Update Options
     * =========================
     */

    function updateOptions(
        questionId,
        questionType
    ) {

        let labels = [];


        if (questionType === 'MCQ') {

            labels =
                ['A', 'B', 'C', 'D'];

        }

        else if (
            questionType === 'TRUE_FALSE'
        ) {

            labels =
                ['A', 'B'];

        }


        const correctOption =
            document.querySelector(
                'input[name="editCorrectOption"]:checked'
            );


        if (!correctOption) {

            throw new Error(
                'Please select correct answer'
            );

        }


        const correctLabel =
            correctOption.value;


        return fetch(
            `/api/questions/${questionId}/options`,
            {

                method: 'DELETE'

            }
        )

        .then(response => {

            if (!response.ok) {

                throw new Error(
                    'Failed to delete old options'
                );

            }

            return response.json();

        })

        .then(() => {

            const requests =
                labels.map(label => {

                    const optionText =
                        document.getElementById(
                            `editOption_${label}`
                        ).value;


                    const option = {

                        optionLabel:
                            label,

                        optionText:
                            optionText,

                        isCorrect:
                            label ===
                            correctLabel

                    };


                    return fetch(
                        `/api/questions/${questionId}/options`,
                        {

                            method: 'POST',

                            headers: {

                                'Content-Type':
                                    'application/json'

                            },

                            body:
                                JSON.stringify(
                                    option
                                )

                        }
                    )

                    .then(response => {

                        if (!response.ok) {

                            throw new Error(
                                `Failed to save option ${label}`
                            );

                        }

                        return response.json();

                    });

                });


            return Promise.all(
                requests
            );

        });

    }


    /*
     * =========================
     * Delete Question
     * =========================
     */

    function deleteQuestion(
        questionId
    ) {

        const confirmDelete =
            confirm(
                "Are you sure you want to delete this question?"
            );


        if (!confirmDelete) {

            return;

        }


        fetch(
            `/api/questions/${questionId}`,
            {

                method: 'DELETE'

            }
        )

        .then(response => {

            if (!response.ok) {

                throw new Error(
                    'Failed to delete question'
                );

            }

            return response.text();

        })

        .then(data => {

            console.log(
                "Delete result:",
                data
            );

            alert(
                "Question deleted successfully!"
            );

            loadQuestions();

        })

        .catch(error => {

            console.error(
                "Error deleting question:",
                error
            );

            alert(
                "Failed to delete question."
            );

        });

    }


    /*
     * =========================
     * Search
     * =========================
     */

    document
        .getElementById(
            'searchInput'
        )
        .addEventListener(
            'input',
            function () {

                const searchText =
                    this.value.toLowerCase();


                const rows =
                    document.querySelectorAll(
                        '#questionTableBody tr'
                    );


                rows.forEach(row => {

                    const questionText =
                        row.innerText.toLowerCase();


                    if (
                        questionText.includes(
                            searchText
                        )
                    ) {

                        row.style.display =
                            '';

                    }

                    else {

                        row.style.display =
                            'none';

                    }

                });

            }
        );


    /*
     * =========================
     * Type Filter
     * =========================
     */

    document
        .getElementById(
            'typeFilter'
        )
        .addEventListener(
            'change',
            function () {

                const selectedType =
                    this.value;


                const rows =
                    document.querySelectorAll(
                        '#questionTableBody tr'
                    );


                rows.forEach(row => {

                    if (!selectedType) {

                        row.style.display =
                            '';

                        return;

                    }


                    const rowText =
                        row.innerText;


                    if (
                        rowText.includes(
                            selectedType
                        )
                    ) {

                        row.style.display =
                            '';

                    }

                    else {

                        row.style.display =
                            'none';

                    }

                });

            }
        );
